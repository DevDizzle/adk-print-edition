% ---- Notes (admonitions) -------------------------------------------------
% A tcolorbox inside a breakable tcolorbox cannot break across pages. Inside a
% note, code blocks drop their own box (the note already has a gray fill).
\newcommand{\flatcode}{%
  \renewenvironment{Shaded}{\par\smallskip}{\par\smallskip}%
  \renewenvironment{plaincode}{\VerbatimEnvironment\begin{plainverb}}{\end{plainverb}}}
\newcommand{\admontitle}[1]{\flatcode\ifx&#1&\else{\sffamily\bfseries\small #1}\par\smallskip\fi}
\newtcolorbox{admon}[1]{enhanced,breakable,colback=black!3,frame hidden,boxrule=0pt,arc=0pt,
  borderline west={2.5pt}{0pt}{black!45},left=10pt,right=8pt,top=5pt,bottom=5pt,
  before skip=10pt,after skip=10pt,before upper={\admontitle{#1}}}
\newtcolorbox{admonstrong}[1]{enhanced,breakable,colback=black!6,frame hidden,boxrule=0pt,arc=0pt,

% ---- Step opener and checkpoint pages --------------------------------------
\newcommand{\stepopener}[2]{%
  \cleardoublepage\thispagestyle{empty}%
  \phantomsection\addcontentsline{toc}{part}{#1: #2}%
  \markboth{#1: #2}{}%
  \vspace*{0.7in}%
  {\sffamily\Large\color{black!60}\MakeUppercase{#1}\par}\vspace{4pt}%
  {\sffamily\bfseries\Huge #2\par}\vspace{8pt}%
  \hrule height 1.2pt\vspace{0.35in}}
\newcommand{\stepchapter}[3]{\noindent\makebox[2.2em][r]{#1}\hspace{0.9em}#2\dotfill\pageref{#3}\par\smallskip}

  {\small #2}\ifx&#3&\else\\{\small #3}\fi\end{minipage}\par\medskip}
\newcommand{\checkitem}[3]{\stepdo{#1}{#2}{#3}}

  You finished the reading for Step #1. #3\par\medskip
  {\sffamily\bfseries Do now}\par\medskip}
 {\par\medskip Date done: \rule{2.2in}{0.4pt}\par
  \end{cpbox}
  \vspace{0.25in}{\sffamily\bfseries Notes}\par\medskip
  \notelines}
